# DLC Labeler — base install.
#
# Everything here is needed to label frames, run MediaPipe, view
# predictions in 2D and 3D, and commit a DeepLabCut training set.
# Training and inference need DeepLabCut itself, which is several GB and
# lives in requirements-dlc.txt — install it with `./setup.sh --with-dlc`
# or the button on the Jobs page, once a machine actually needs to train.

# Web app
fastapi>=0.100.0
uvicorn>=0.20.0
python-multipart>=0.0.5
pydantic>=2.0.0

# Video, frames and geometry
opencv-python-headless>=4.8.0
numpy>=1.20.0,<2.0
pyyaml>=5.0

# Bundled ffmpeg, so nothing has to be installed system-wide
imageio-ffmpeg>=0.4.9

# Hand landmarks.
#
# Pinned below 0.10.19: that release removed the `mediapipe.solutions`
# API this app detects hands with.  Lifting the pin means rewriting
# services/mediapipe_prelabel.py against the Tasks API, not just
# bumping a number.
mediapipe>=0.10.0,<0.10.19
protobuf>=3.20,<5

# Frame selection.
#
# DeepLabCut's own k-means frame picker is MiniBatchKMeans from
# scikit-learn.  It is in the base tier because selecting frames to
# label (and to send out for labeling) has to work on a machine that
# will never train.
scikit-learn>=1.0
