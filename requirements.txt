# DLC Labeler.
#
# This app labels frames and writes DeepLabCut training sets. It does not
# run MediaPipe and it does not run DeepLabCut, so neither is a
# dependency — which is why the install is small and why it is not
# capped at Python 3.12 the way anything importing mediapipe is.

# Web app
fastapi>=0.100.0
uvicorn>=0.20.0
python-multipart>=0.0.5
pydantic>=2.0.0

# Video, frames and geometry
opencv-python-headless>=4.8.0
numpy>=1.20.0,<2.0
pyyaml>=5.0

# Bundled ffmpeg, so nothing has to be installed system-wide
imageio-ffmpeg>=0.4.9

# Frame selection: DeepLabCut's own k-means frame picker is
# MiniBatchKMeans from scikit-learn, so this app can reproduce the
# selection without DeepLabCut installed.
scikit-learn>=1.0
